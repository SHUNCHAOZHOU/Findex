import numpy as np

FAST = 10
SLOW = 30
MOM = 10
BREAK = 10
ATR_N = 14
TRAIL = 0.10

def _ema(vals, n):
    if len(vals) < n:
        return None
    k = 2.0 / (n + 1.0)
    e = vals[0]
    for v in vals[1:]:
        e = v * k + e * (1 - k)
    return e

def _atr(high, low, close, n):
    if len(close) < n + 1:
        return None
    trs = []
    for i in range(len(close) - n, len(close)):
        tr = max(high[i] - low[i], abs(high[i] - close[i-1]), abs(low[i] - close[i-1]))
        trs.append(tr)
    return sum(trs) / len(trs)

def decide(bars, account):
    close = bars["close"]
    high = bars["high"]
    low = bars["low"]
    n = len(close)
    if n < SLOW + 2:
        return "HOLD"

    fast = _ema(close[-FAST:], FAST)
    slow = _ema(close[-SLOW:], SLOW)
    if fast is None or slow is None:
        return "HOLD"

    price = close[-1]
    mom = price - close[-1 - MOM] if n > MOM else 0.0
    recent_low = min(low[-BREAK:])

    holding = account.get("position", 0) > 0

    if not holding:
        if fast > slow and price > slow and mom > 0:
            return ("BUY", "uptrend confirmed")
        return "HOLD"
    else:
        entry = account.get("entry_price")
        if fast < slow or price < recent_low:
            return ("SELL", "trend flip or 10-day low break")
        if entry is not None and price < entry * (1 - TRAIL):
            return ("SELL", "trailing stop")
        return "HOLD"
