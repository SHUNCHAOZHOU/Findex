import math

FAST = 20
SLOW = 50
MOM = 10


def _sma(vals, n):
    if len(vals) < n:
        return None
    return sum(vals[-n:]) / n


def decide(bars, account):
    closes = bars["close"]
    n = len(closes)
    if n < SLOW + 2:
        return "HOLD"

    price = closes[-1]
    fast = _sma(closes, FAST)
    fast_prev = _sma(closes[:-1], FAST)
    slow = _sma(closes, SLOW)
    slow_prev = _sma(closes[:-1], SLOW)
    mom = _sma(closes, MOM)

    if fast is None or fast_prev is None or slow is None or slow_prev is None or mom is None:
        return "HOLD"

    pos = account.get("position", 0)

    if pos == 0:
        if price > fast and fast > fast_prev and price > mom and price > slow:
            return ("BUY", "price>rising SMA20, above momentum and above SMA50")
        return "HOLD"
    else:
        # Regime exit: only leave when the long-term trend has actually broken.
        if price < slow and slow <= slow_prev:
            return ("SELL", "below SMA50 with trend no longer rising")
        return "HOLD"

import math

FAST = 20
MOM = 10
ATR_N = 14
ATR_MULT = 2.5


def _sma(vals, n):
    if len(vals) < n:
        return None
    return sum(vals[-n:]) / n


def _atr(bars, n):
    highs = bars["high"]
    lows = bars["low"]
    closes = bars["close"]
    if len(closes) < n + 1:
        return None
    trs = []
    for i in range(len(closes) - n, len(closes)):
        tr = max(highs[i] - lows[i], abs(highs[i] - closes[i - 1]), abs(lows[i] - closes[i - 1]))
        trs.append(tr)
    return sum(trs) / n


def decide(bars, account):
    closes = bars["close"]
    n = len(closes)
    if n < FAST + MOM + 2:
        return "HOLD"

    price = closes[-1]
    fast = _sma(closes, FAST)
    fast_prev = _sma(closes[:-1], FAST)
    mom = _sma(closes, MOM)
    atr = _atr(bars, ATR_N)

    if fast is None or fast_prev is None or mom is None or atr is None:
        return "HOLD"

    pos = account.get("position", 0)

    if pos == 0:
        if price > fast and fast > fast_prev and price > mom:
            return ("BUY", "price>rising SMA20 and above momentum")
        return "HOLD"
    else:
        highest = max(closes[-FAST:])
        trail = highest - ATR_MULT * atr
        # Regime-aware exit: only bail on SMA20 break if the trend has also stopped rising.
        if price < trail:
            return ("SELL", "ATR trailing stop")
        if price < fast and fast <= fast_prev:
            return ("SELL", "below SMA20 with trend no longer rising")
        return "HOLD"

Trend-following long-only strategy: hold when price is above a rising long-term moving average and momentum is positive, exit when trend breaks down. Uses a 50-day SMA as the regime filter plus a faster 20-day SMA crossover for entries, with an ATR-based trailing stop from entry price to protect gains. import math

FAST = 20
SLOW = 50
ATR_N = 14
ATR_MULT = 3.0


def _sma(vals, n):
    if len(vals) < n:
        return None
    return sum(vals[-n:]) / n


def _atr(bars, n):
    highs = bars["high"]
    lows = bars["low"]
    closes = bars["close"]
    if len(closes) < n + 1:
        return None
    trs = []
    for i in range(len(closes) - n, len(closes)):
        tr = max(highs[i] - lows[i], abs(highs[i] - closes[i - 1]), abs(lows[i] - closes[i - 1]))
        trs.append(tr)
    return sum(trs) / n


def decide(bars, account):
    closes = bars["close"]
    n = len(closes)
    if n < SLOW + 2:
        return "HOLD"

    price = closes[-1]
    fast = _sma(closes, FAST)
    slow = _sma(closes, SLOW)
    slow_prev = _sma(closes[:-5], SLOW) if n >= SLOW + 5 else slow
    atr = _atr(bars, ATR_N)

    if fast is None or slow is None or atr is None:
        return "HOLD"

    trend_up = price > slow and slow >= slow_prev
    momentum = fast > slow

    pos = account.get("position", 0)
    entry = account.get("entry_price")

    if pos == 0:
        if trend_up and momentum and price > fast:
            return ("BUY", "uptrend: price>slow SMA, fast>slow")
        return "HOLD"
    else:
        stop = None
        if entry is not None:
            stop = entry - ATR_MULT * atr
        if price < slow or (stop is not None and price < stop):
            return ("SELL", "trend break or ATR stop")
        if fast < slow and price < fast:
            return ("SELL", "momentum flip")
        return "HOLD"
