import numpy as np

FAST = 20
SLOW = 50
TREND = 200
RSI_N = 14
RSI_OVERBOUGHT = 85


def _ema(values, period):
    if len(values) < period:
        return None
    k = 2.0 / (period + 1.0)
    ema = values[0]
    for v in values[1:]:
        ema = v * k + ema * (1 - k)
    return ema


def _sma(values, period):
    if len(values) < period:
        return None
    return sum(values[-period:]) / period


def _rsi(closes, period):
    if len(closes) < period + 1:
        return None
    gains = 0.0
    losses = 0.0
    for i in range(len(closes) - period, len(closes)):
        diff = closes[i] - closes[i - 1]
        if diff >= 0:
            gains += diff
        else:
            losses -= diff
    avg_gain = gains / period
    avg_loss = losses / period
    if avg_loss == 0:
        return 100.0
    rs = avg_gain / avg_loss
    return 100.0 - 100.0 / (1.0 + rs)


def decide(bars, account):
    closes = list(bars["close"])
    n = len(closes)
    if n < SLOW + 2:
        return "HOLD"

    price = closes[-1]
    fast = _ema(closes, FAST)
    slow = _ema(closes, SLOW)
    trend = _sma(closes, TREND) if n >= TREND else None
    rsi = _rsi(closes, RSI_N)

    if fast is None or slow is None:
        return "HOLD"

    holding = account.get("position", 0) > 0

    trend_ok = (trend is None) or (price > trend)

    # Hysteresis: require a small margin for crossover signals
    margin = 0.002 * slow
    bullish = (fast - slow) > margin and trend_ok
    bearish = (slow - fast) > margin

    if not holding:
        if bullish and (rsi is None or rsi < RSI_OVERBOUGHT):
            return ("BUY", "EMA%d>EMA%d and trend up" % (FAST, SLOW))
        return "HOLD"
    else:
        if bearish:
            return ("SELL", "EMA%d<EMA%d momentum breakdown" % (FAST, SLOW))
        if trend is not None and price < trend * 0.95:
            return ("SELL", "below long trend")
        return "HOLD"
