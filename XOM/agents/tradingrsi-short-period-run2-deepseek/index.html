First version: a trend-following long-only strategy for XOM using a 50/200-day moving-average regime filter plus a shorter momentum/RSI pullback entry. Buy when price is above the 200-day average and momentum (20-day return) is positive, or when a short-term oversold dip occurs within an uptrend; sell when price breaks below the 50-day average or momentum turns negative. This balances capturing XOM's cyclical uptrends while avoiding deep drawdowns, and I'd revise if whipsaws in choppy markets cause excessive churn or if the trend filter lags too much.
import math

def _sma(vals, n):
    if len(vals) < n:
        return None
    return sum(vals[-n:]) / n

def _rsi(closes, n=14):
    if len(closes) < n + 1:
        return None
    gains = 0.0
    losses = 0.0
    for i in range(len(closes) - n, len(closes)):
        d = closes[i] - closes[i - 1]
        if d >= 0:
            gains += d
        else:
            losses -= d
    if losses == 0:
        return 100.0
    rs = gains / losses
    return 100.0 - 100.0 / (1.0 + rs)

def decide(bars, account):
    closes = bars["close"]
    n = len(closes)
    if n < 60:
        return "HOLD"
    price = closes[-1]
    sma50 = _sma(closes, 50)
    sma200 = _sma(closes, 200) if n >= 200 else None
    sma20 = _sma(closes, 20)
    rsi = _rsi(closes, 14)
    mom20 = (price / closes[-21] - 1.0) if n >= 21 else 0.0
    mom60 = (price / closes[-61] - 1.0) if n >= 61 else 0.0

    holding = account.get("position", 0) > 0

    # Long-term regime: above 200d if available, else above 50d
    if sma200 is not None:
        uptrend = price > sma200 and sma50 > sma200
    else:
        uptrend = price > sma50

    if not holding:
        # Entry: uptrend with positive momentum, or oversold pullback in uptrend
        if uptrend and mom20 > 0 and mom60 > -0.02:
            return ("BUY", "uptrend+momentum")
        if uptrend and rsi is not None and rsi < 40 and price > sma50 * 0.97:
            return ("BUY", "uptrend pullback")
        return "HOLD"
    else:
        # Exit: trend break or momentum reversal
        if sma50 is not None and price < sma50 * 0.98:
            return ("SELL", "below 50dma")
        if mom20 < -0.06:
            return ("SELL", "momentum reversal")
        if sma200 is not None and price < sma200:
            return ("SELL", "below 200dma")
        return "HOLD"
