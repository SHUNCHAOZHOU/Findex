Trend-following with a long-only moving-average crossover: hold when the fast EMA is above the slow EMA and price is above a longer trend filter, exit when momentum breaks down. This captures the strong uptrend visible in the recent bars while avoiding deep drawdowns. I'd change it if whipsaws dominate (add a buffer/hysteresis band) or if a volatility filter improves risk-adjusted returns.
import math

FAST = 20
SLOW = 50
TREND = 200
ATR_WIN = 14


def _ema(vals, n):
    k = 2.0 / (n + 1.0)
    e = vals[0]
    for v in vals[1:]:
        e = v * k + e * (1.0 - k)
    return e


def _sma(vals, n):
    return sum(vals[-n:]) / float(n)


def _atr(bars, n):
    highs = bars["high"]
    lows = bars["low"]
    closes = bars["close"]
    m = len(closes)
    if m < n + 1:
        return None
    trs = []
    for i in range(m - n, m):
        tr = max(highs[i] - lows[i],
                 abs(highs[i] - closes[i - 1]),
                 abs(lows[i] - closes[i - 1]))
        trs.append(tr)
    return sum(trs) / float(n)


def decide(bars, account):
    closes = bars["close"]
    n = len(closes)
    if n < SLOW + 2:
        return "HOLD"

    price = closes[-1]
    fast = _ema(closes[-FAST * 3:], FAST)
    slow = _ema(closes[-SLOW * 3:], SLOW)

    trend_ok = True
    if n >= TREND:
        trend_ok = price > _sma(closes, TREND)

    atr = _atr(bars, ATR_WIN)
    vol_ok = True
    if atr is not None and price > 0:
        vol_ok = (atr / price) < 0.06

    holding = account.get("position", 0) > 0

    if not holding:
        if fast > slow and trend_ok and vol_ok and price > slow:
            return ("BUY", "EMA%d>EMA%d and above trend" % (FAST, SLOW))
        return "HOLD"
    else:
        if fast < slow or not trend_ok:
            return ("SELL", "trend break: EMA%d<EMA%d or below SMA%d" % (FAST, SLOW, TREND))
        return "HOLD"
