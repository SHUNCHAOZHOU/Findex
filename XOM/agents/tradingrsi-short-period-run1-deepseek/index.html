import statistics

SMA_FAST = 20
SMA_SLOW = 50
SMA_LONG = 100
ATR_WIN = 14
STOP_ATR = 5.0
MOM_WIN = 20
MOM_FAST = 10


def _sma(vals, n):
    if len(vals) < n:
        return None
    return sum(vals[-n:]) / n


def _atr(bars, n):
    highs = bars["high"]
    lows = bars["low"]
    closes = bars["close"]
    if len(closes) < n + 1:
        return None
    trs = []
    for i in range(len(closes) - n, len(closes)):
        tr = max(highs[i] - lows[i], abs(highs[i] - closes[i - 1]), abs(lows[i] - closes[i - 1]))
        trs.append(tr)
    return sum(trs) / n


def decide(bars, account):
    closes = bars["close"]
    n = len(closes)
    if n < SMA_SLOW + 12:
        return "HOLD"

    price = closes[-1]
    sma_fast = _sma(closes, SMA_FAST)
    sma_fast_prev = sum(closes[-SMA_FAST - 5:-5]) / SMA_FAST
    sma_slow = _sma(closes, SMA_SLOW)
    sma_slow_prev = sum(closes[-SMA_SLOW - 10:-10]) / SMA_SLOW
    sma_long = _sma(closes, SMA_LONG)
    atr = _atr(bars, ATR_WIN)
    mom = (price / closes[-MOM_WIN - 1] - 1.0) if n > MOM_WIN else 0.0
    mom_fast = (price / closes[-MOM_FAST - 1] - 1.0) if n > MOM_FAST else 0.0

    holding = account.get("position", 0) > 0
    entry = account.get("entry_price")

    if holding:
        if sma_slow is not None and price < sma_slow:
            return ("SELL", "close below 50d SMA")
        if entry is not None and atr is not None and price < entry - STOP_ATR * atr:
            return ("SELL", "wide ATR trailing stop")
        return "HOLD"

    if sma_slow is None or sma_fast is None:
        return "HOLD"
    rising = sma_slow > sma_slow_prev
    long_uptrend = sma_long is not None and price > sma_long and mom > 0
    fast_uptrend = (sma_fast is not None and sma_fast_prev is not None
                    and price > sma_fast and sma_fast > sma_fast_prev and mom_fast > 0)
    if (price > sma_slow and rising) or long_uptrend or fast_uptrend:
        return ("BUY", "uptrend confirmed")
    return "HOLD"

Trend-following long-only strategy for XOM: buy when price is above a rising 50-day SMA and momentum (20-day return) is positive, confirming an uptrend; exit when price closes below the 50-day SMA or a faster 10-day SMA crosses below the 50-day, whichever comes first. ATR-based trailing stop from entry protects against sharp reversals like the mid-May spike-and-fade. import statistics

SMA_FAST = 10
SMA_SLOW = 50
MOM_WIN = 20
ATR_WIN = 14
STOP_ATR = 3.0


def _sma(vals, n):
    if len(vals) < n:
        return None
    return sum(vals[-n:]) / n


def _atr(bars, n):
    highs = bars["high"]
    lows = bars["low"]
    closes = bars["close"]
    if len(closes) < n + 1:
        return None
    trs = []
    for i in range(len(closes) - n, len(closes)):
        tr = max(highs[i] - lows[i], abs(highs[i] - closes[i - 1]), abs(lows[i] - closes[i - 1]))
        trs.append(tr)
    return sum(trs) / n


def decide(bars, account):
    closes = bars["close"]
    n = len(closes)
    if n < SMA_SLOW + 2:
        return "HOLD"

    price = closes[-1]
    sma_fast = _sma(closes, SMA_FAST)
    sma_slow = _sma(closes, SMA_SLOW)
    sma_slow_prev = sum(closes[-SMA_SLOW - 1:-1]) / SMA_SLOW
    mom = price / closes[-MOM_WIN - 1] - 1.0 if n > MOM_WIN else 0.0
    atr = _atr(bars, ATR_WIN)

    holding = account.get("position", 0) > 0
    entry = account.get("entry_price")

    if holding:
        # Exit on trend break
        if price < sma_slow:
            return ("SELL", "close below 50d SMA")
        if sma_fast is not None and sma_fast < sma_slow:
            return ("SELL", "10d SMA below 50d SMA")
        # ATR trailing stop from entry
        if entry is not None and atr is not None and price < entry - STOP_ATR * atr:
            return ("SELL", "ATR trailing stop")
        return "HOLD"

    # Flat: look for uptrend entry
    if sma_slow is None or sma_fast is None:
        return "HOLD"
    uptrend = price > sma_slow and sma_slow > sma_slow_prev and sma_fast > sma_slow
    if uptrend and mom > 0.0:
        return ("BUY", "uptrend with positive momentum")
    return "HOLD"
