import math

FAST = 20
SLOW = 50
LONG = 100
ATR_N = 14
ATR_MULT = 4.0


def _sma(vals, n):
    if len(vals) < n:
        return None
    return sum(vals[-n:]) / n


def _atr(highs, lows, closes, n):
    if len(closes) < n + 1:
        return None
    trs = []
    for i in range(len(closes) - n, len(closes)):
        tr = max(highs[i] - lows[i],
                 abs(highs[i] - closes[i - 1]),
                 abs(lows[i] - closes[i - 1]))
        trs.append(tr)
    return sum(trs) / n


def decide(bars, account):
    closes = bars["close"]
    highs = bars["high"]
    lows = bars["low"]
    n = len(closes)
    if n < SLOW + 2:
        return "HOLD"

    price = closes[-1]
    fast = _sma(closes, FAST)
    slow = _sma(closes, SLOW)
    long_ma = _sma(closes, LONG)
    atr = _atr(highs, lows, closes, ATR_N)

    pos = account.get("position", 0)
    entry = account.get("entry_price")

    if pos > 0:
        # hard catastrophic stop only
        if entry is not None and atr is not None:
            if price < entry - ATR_MULT * atr:
                return ("SELL", "wide ATR stop")
        # exit only on a genuine trend break: below 50-day AND 20 below 50
        if slow is not None and fast is not None:
            if price < slow and fast < slow:
                return ("SELL", "trend break")
        return "HOLD"

    if fast is None or slow is None:
        return "HOLD"
    # regime: stay invested when above long MA or short-term trend up
    above_long = long_ma is not None and price > long_ma
    uptrend = fast > slow
    if not (above_long or uptrend):
        return "HOLD"
    if price < slow * 0.95:
        return "HOLD"
    return ("BUY", "regime long")

import math

FAST = 20
SLOW = 50
LONG = 200
RSI_N = 14
ATR_N = 14
ATR_MULT = 3.0


def _sma(vals, n):
    if len(vals) < n:
        return None
    return sum(vals[-n:]) / n


def _rsi(closes, n):
    if len(closes) < n + 1:
        return None
    gains = 0.0
    losses = 0.0
    for i in range(len(closes) - n, len(closes)):
        d = closes[i] - closes[i - 1]
        if d >= 0:
            gains += d
        else:
            losses -= d
    if losses == 0:
        return 100.0
    rs = (gains / n) / (losses / n)
    return 100.0 - 100.0 / (1.0 + rs)


def _atr(highs, lows, closes, n):
    if len(closes) < n + 1:
        return None
    trs = []
    for i in range(len(closes) - n, len(closes)):
        tr = max(highs[i] - lows[i],
                 abs(highs[i] - closes[i - 1]),
                 abs(lows[i] - closes[i - 1]))
        trs.append(tr)
    return sum(trs) / n


def decide(bars, account):
    closes = bars["close"]
    highs = bars["high"]
    lows = bars["low"]
    n = len(closes)
    if n < SLOW + 2:
        return "HOLD"

    price = closes[-1]
    fast = _sma(closes, FAST)
    slow = _sma(closes, SLOW)
    long_ma = _sma(closes, LONG) if n >= LONG else None
    rsi = _rsi(closes, RSI_N)
    atr = _atr(highs, lows, closes, ATR_N)

    pos = account.get("position", 0)
    entry = account.get("entry_price")

    # Exit logic when holding
    if pos > 0:
        # trailing stop
        if entry is not None and atr is not None:
            if price < entry - ATR_MULT * atr:
                return ("SELL", "trailing stop hit")
        # trend break
        if slow is not None and price < slow:
            return ("SELL", "price below 50-day MA")
        if fast is not None and slow is not None and fast < slow:
            return ("SELL", "fast MA crossed below slow MA")
        if rsi is not None and rsi > 80:
            return ("SELL", "RSI overbought")
        return "HOLD"

    # Entry logic when flat
    if fast is None or slow is None:
        return "HOLD"
    if fast <= slow:
        return "HOLD"
    if long_ma is not None and price < long_ma:
        return "HOLD"
    if rsi is not None and rsi > 70:
        return "HOLD"
    if rsi is not None and rsi < 45:
        return "HOLD"
    # require some upward momentum
    if n >= FAST + 5 and closes[-1] <= closes[-FAST]:
        return "HOLD"
    return ("BUY", "uptrend with momentum")
