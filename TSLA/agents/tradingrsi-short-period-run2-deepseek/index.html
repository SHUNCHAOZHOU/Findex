Trend-following with a fast/slow EMA crossover plus a momentum filter, designed to capture TSLA's large directional swings while staying long-only. I use a 20/50 EMA crossover for entries and exits, with an ATR-based trailing stop from entry to protect gains and avoid deep drawdowns. A 200-day trend filter keeps us out of sustained downtrends. I would change it if whipsaws dominate (tighten filters) or if the stock becomes mean-reverting (switch to RSI/mean-reversion).
import math

FAST = 20
SLOW = 50
TREND = 200
ATR_N = 14
ATR_MULT = 3.0


def _ema(vals, n):
    if len(vals) < n:
        return None
    k = 2.0 / (n + 1.0)
    e = sum(vals[:n]) / n
    for v in vals[n:]:
        e = v * k + e * (1 - k)
    return e


def _atr(bars, n):
    highs = bars["high"]
    lows = bars["low"]
    closes = bars["close"]
    if len(closes) < n + 1:
        return None
    trs = []
    for i in range(len(closes) - n, len(closes)):
        tr = max(highs[i] - lows[i], abs(highs[i] - closes[i - 1]), abs(lows[i] - closes[i - 1]))
        trs.append(tr)
    return sum(trs) / n


def decide(bars, account):
    closes = bars["close"]
    n = len(closes)
    if n < SLOW + 2:
        return "HOLD"

    price = closes[-1]
    fast = _ema(closes, FAST)
    slow = _ema(closes, SLOW)
    if fast is None or slow is None:
        return "HOLD"

    trend_ok = True
    if n >= TREND:
        sma_trend = sum(closes[-TREND:]) / TREND
        trend_ok = price > sma_trend

    pos = account.get("position", 0)
    entry = account.get("entry_price")

    if pos == 0:
        # Entry: fast above slow, price above slow, and trend filter ok
        if fast > slow and price > slow and trend_ok:
            # require recent momentum: price above 10-day ago close
            if n >= 11 and price > closes[-11]:
                return ("BUY", "EMA%d>EMA%d uptrend, price above trend" % (FAST, SLOW))
        return "HOLD"
    else:
        # Exit conditions
        if fast < slow:
            return ("SELL", "EMA%d crossed below EMA%d" % (FAST, SLOW))
        if not trend_ok:
            return ("SELL", "price below %d-day trend" % TREND)
        atr = _atr(bars, ATR_N)
        if atr is not None and entry is not None:
            if price < entry - ATR_MULT * atr:
                return ("SELL", "ATR trailing stop hit")
        return "HOLD"
