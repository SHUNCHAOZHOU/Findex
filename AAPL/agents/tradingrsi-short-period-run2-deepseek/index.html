import math

TREND_N = 100
EXIT_DD = 0.10


def _sma(vals, n):
    if len(vals) < n:
        return None
    return sum(vals[-n:]) / n


def decide(bars, account):
    closes = bars["close"]
    if len(closes) < TREND_N + 2:
        return "HOLD"

    price = closes[-1]
    trend = _sma(closes, TREND_N)
    trend_prev = sum(closes[-TREND_N-1:-1]) / TREND_N
    if trend is None:
        return "HOLD"

    pos = account.get("position", 0)
    entry = account.get("entry_price")

    rising = trend > trend_prev
    above = price > trend

    if pos == 0:
        if above and rising:
            return ("BUY", "price above rising 100d SMA")
        return "HOLD"
    else:
        if entry is not None and price < entry * (1.0 - EXIT_DD):
            return ("SELL", "drawdown stop from entry")
        if price < trend:
            return ("SELL", "price below 100d SMA")
        return "HOLD"

Trend-following long-only strategy: hold when price is above a rising medium-term moving average and momentum is positive, exit when trend breaks down. Uses a 50-day SMA as the primary regime filter plus a faster 20-day SMA for entries/exits, with an ATR-based trailing stop from entry to protect gains. I'd change it if whipsaws dominate or if drawdowns during choppy periods exceed what a simple trend filter can tolerate.
import math

FAST = 20
SLOW = 50
ATR_N = 14
ATR_MULT = 3.0

def _sma(vals, n):
    if len(vals) < n:
        return None
    return sum(vals[-n:]) / n

def _atr(bars, n):
    highs = bars["high"]
    lows = bars["low"]
    closes = bars["close"]
    if len(closes) < n + 1:
        return None
    trs = []
    for i in range(len(closes) - n, len(closes)):
        tr = max(highs[i] - lows[i], abs(highs[i] - closes[i-1]), abs(lows[i] - closes[i-1]))
        trs.append(tr)
    return sum(trs) / len(trs)

def decide(bars, account):
    closes = bars["close"]
    if len(closes) < SLOW + 2:
        return "HOLD"

    price = closes[-1]
    fast = _sma(closes, FAST)
    slow = _sma(closes, SLOW)
    slow_prev = sum(closes[-SLOW-1:-1]) / SLOW
    atr = _atr(bars, ATR_N)

    if fast is None or slow is None or atr is None:
        return "HOLD"

    pos = account.get("position", 0)
    entry = account.get("entry_price")

    trend_up = price > slow and slow > slow_prev
    momentum_ok = fast > slow

    if pos == 0:
        if trend_up and momentum_ok and price > fast:
            return ("BUY", "uptrend: price>slow SMA, fast>slow, rising slow")
        return "HOLD"
    else:
        if entry is not None and price < entry - ATR_MULT * atr:
            return ("SELL", "ATR trailing stop hit")
        if price < slow or fast < slow:
            return ("SELL", "trend breakdown: price<slow or fast<slow")
        return "HOLD"
