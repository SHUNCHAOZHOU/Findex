import math

TREND = 100
ATR_N = 14
ATR_MULT = 4.0
BAND = 0.01


def _ema(vals, n):
    if len(vals) < n:
        return None
    k = 2.0 / (n + 1.0)
    e = sum(vals[:n]) / n
    for v in vals[n:]:
        e = v * k + e * (1 - k)
    return e


def _atr(bars, n):
    highs = bars["high"]
    lows = bars["low"]
    closes = bars["close"]
    if len(closes) < n + 1:
        return None
    trs = []
    for i in range(len(closes) - n, len(closes)):
        tr = max(highs[i] - lows[i], abs(highs[i] - closes[i - 1]), abs(lows[i] - closes[i - 1]))
        trs.append(tr)
    return sum(trs) / n


def decide(bars, account):
    closes = bars["close"]
    n = len(closes)
    if n < TREND + 2:
        return "HOLD"

    price = closes[-1]
    trend = _ema(closes, TREND)
    trend_prev = _ema(closes[:-1], TREND)
    atr = _atr(bars, ATR_N)

    if trend is None or trend_prev is None:
        return "HOLD"

    pos = account.get("position", 0)

    # Regime: price above a rising long EMA
    above = price > trend * (1 + BAND)
    rising = trend > trend_prev
    bullish = above and rising
    bearish = price < trend * (1 - BAND)

    if pos == 0:
        if bullish:
            return ("BUY", "price above rising EMA%d" % TREND)
        return "HOLD"

    if bearish:
        return ("SELL", "price below EMA%d" % TREND)

    # Chandelier stop from highest close since entry
    if atr is not None:
        lookback = min(n, 60)
        recent_high = max(closes[-lookback:])
        stop = recent_high - ATR_MULT * atr
        if price < stop:
            return ("SELL", "chandelier stop")

    return "HOLD"

I'll start with a trend-following strategy that combines a fast/slow EMA crossover with a longer-term trend filter and an ATR-based trailing stop, since MSFT is a large-cap that trends well and the recent data shows a sharp breakout. The idea is to stay long while the medium-term trend is up, exit on momentum deterioration or a stop, and avoid whipsaw with a confirmation band. I'd change it if backtests show too much churn or if the trailing stop cuts winners too early.
import math

FAST = 20
SLOW = 50
TREND = 200
ATR_N = 14
ATR_MULT = 3.0
BAND = 0.005


def _ema(vals, n):
    if len(vals) < n:
        return None
    k = 2.0 / (n + 1.0)
    e = sum(vals[:n]) / n
    for v in vals[n:]:
        e = v * k + e * (1 - k)
    return e


def _atr(bars, n):
    highs = bars["high"]
    lows = bars["low"]
    closes = bars["close"]
    if len(closes) < n + 1:
        return None
    trs = []
    for i in range(len(closes) - n, len(closes)):
        tr = max(highs[i] - lows[i], abs(highs[i] - closes[i - 1]), abs(lows[i] - closes[i - 1]))
        trs.append(tr)
    return sum(trs) / n


def decide(bars, account):
    closes = bars["close"]
    n = len(closes)
    if n < SLOW + 2:
        return "HOLD"

    price = closes[-1]
    fast = _ema(closes, FAST)
    slow = _ema(closes, SLOW)
    trend = _ema(closes, TREND) if n >= TREND else None
    atr = _atr(bars, ATR_N)

    if fast is None or slow is None:
        return "HOLD"

    pos = account.get("position", 0)
    entry = account.get("entry_price")

    # Trend filter: require price above long EMA when available
    trend_ok = (trend is None) or (price > trend)

    bullish = fast > slow * (1 + BAND) and trend_ok
    bearish = fast < slow * (1 - BAND)

    if pos == 0:
        if bullish:
            return ("BUY", "EMA%d>EMA%d and above trend" % (FAST, SLOW))
        return "HOLD"

    # Holding: exit on bearish crossover or trailing stop
    if bearish:
        return ("SELL", "EMA%d<EMA%d" % (FAST, SLOW))

    if entry is not None and atr is not None:
        stop = entry - ATR_MULT * atr
        if price < stop:
            return ("SELL", "ATR trailing stop")

    return "HOLD"
