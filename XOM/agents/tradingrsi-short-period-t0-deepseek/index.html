import math

FAST = 50
SLOW = 200
RSI_N = 14

def _sma(vals, n):
    if len(vals) < n:
        return None
    return sum(vals[-n:]) / n

def _rsi(closes, n=RSI_N):
    if len(closes) < n + 1:
        return None
    gains = 0.0
    losses = 0.0
    for i in range(len(closes) - n, len(closes)):
        d = closes[i] - closes[i - 1]
        if d >= 0:
            gains += d
        else:
            losses -= d
    if losses == 0:
        return 100.0
    rs = (gains / n) / (losses / n)
    return 100.0 - 100.0 / (1.0 + rs)

def decide(bars, account):
    closes = bars["close"]
    n = len(closes)
    if n < SLOW + 5:
        return "HOLD"
    price = closes[-1]
    sma_fast = _sma(closes, FAST)
    sma_slow = _sma(closes, SLOW)
    rsi = _rsi(closes)
    if sma_fast is None or sma_slow is None or rsi is None:
        return "HOLD"

    pos = account.get("position", 0)
    entry = account.get("entry_price")

    # Long-term uptrend regime: price above the 200-day SMA.
    uptrend = price > sma_slow
    # Slow exit: only leave when the long-term trend actually breaks.
    trend_break = price < sma_slow and sma_fast < sma_slow

    if pos == 0:
        if uptrend:
            return ("BUY", "price %.2f above 200SMA %.2f" % (price, sma_slow))
        return "HOLD"
    else:
        if trend_break:
            return ("SELL", "trend break: price %.2f below 200SMA %.2f" % (price, sma_slow))
        if rsi > 82.0 and entry is not None and price > entry * 1.25:
            return ("SELL", "overbought RSI %.1f after strong gain" % rsi)
        return "HOLD"

First version: a trend-following long-only strategy using a 50/200-day moving-average regime filter plus a shorter momentum/RSI pullback entry. Buy when price is above the 200-day SMA and the 50-day SMA is rising, with a mild pullback (RSI below ~55) to avoid chasing; sell when price breaks below the 50-day SMA or RSI gets overbought and momentum fades. I'd change it if whipsaws dominate or if a simpler buy-and-hold benchmark beats it on risk-adjusted return.
import math

FAST = 50
SLOW = 200
RSI_N = 14

def _sma(vals, n):
    if len(vals) < n:
        return None
    return sum(vals[-n:]) / n

def _rsi(closes, n=RSI_N):
    if len(closes) < n + 1:
        return None
    gains = 0.0
    losses = 0.0
    for i in range(len(closes) - n, len(closes)):
        d = closes[i] - closes[i - 1]
        if d >= 0:
            gains += d
        else:
            losses -= d
    if losses == 0:
        return 100.0
    rs = (gains / n) / (losses / n)
    return 100.0 - 100.0 / (1.0 + rs)

def decide(bars, account):
    closes = bars["close"]
    n = len(closes)
    if n < SLOW + 5:
        return "HOLD"
    price = closes[-1]
    sma_fast = _sma(closes, FAST)
    sma_slow = _sma(closes, SLOW)
    sma_fast_prev = _sma(closes[:-5], FAST)
    rsi = _rsi(closes)
    if sma_fast is None or sma_slow is None or sma_fast_prev is None or rsi is None:
        return "HOLD"

    pos = account.get("position", 0)
    entry = account.get("entry_price")

    uptrend = price > sma_slow and sma_fast > sma_slow and sma_fast > sma_fast_prev
    downtrend = price < sma_fast or sma_fast < sma_slow

    if pos == 0:
        if uptrend and rsi < 60.0:
            return ("BUY", "uptrend with RSI %.1f pullback" % rsi)
        return "HOLD"
    else:
        if downtrend:
            return ("SELL", "trend break: price %.2f vs fast %.2f" % (price, sma_fast))
        if rsi > 78.0 and entry is not None and price > entry * 1.15:
            return ("SELL", "overbought RSI %.1f after strong gain" % rsi)
        return "HOLD"
