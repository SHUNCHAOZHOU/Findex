import math

TREND = 100
ATR_N = 14
ATR_MULT = 4.5
BUF = 0.01


def _ema(vals, n):
    if len(vals) < n:
        return None
    k = 2.0 / (n + 1.0)
    e = sum(vals[:n]) / n
    for v in vals[n:]:
        e = v * k + e * (1 - k)
    return e


def _atr(bars, n):
    highs = bars["high"]
    lows = bars["low"]
    closes = bars["close"]
    if len(closes) < n + 1:
        return None
    trs = []
    for i in range(len(closes) - n, len(closes)):
        tr = max(highs[i] - lows[i], abs(highs[i] - closes[i-1]), abs(lows[i] - closes[i-1]))
        trs.append(tr)
    return sum(trs) / len(trs)


def decide(bars, account):
    closes = bars["close"]
    n = len(closes)
    if n < TREND + 2:
        return "HOLD"

    trend = _ema(closes, TREND)
    if trend is None:
        return "HOLD"

    price = closes[-1]
    atr = _atr(bars, ATR_N)

    pos = account.get("position", 0)
    entry = account.get("entry_price")

    if pos == 0:
        if price > trend * (1.0 + BUF):
            return ("BUY", "price above EMA%d by buffer" % TREND)
        return "HOLD"
    else:
        if price < trend * (1.0 - BUF):
            return ("SELL", "price below EMA%d by buffer" % TREND)
        if entry is not None and atr is not None:
            peak = max(closes[-min(n, 60):])
            if price < peak - ATR_MULT * atr:
                return ("SELL", "ATR trailing stop from peak")
        return "HOLD"

Trend-following with a fast/slow EMA crossover plus a momentum filter, designed to capture TSLA's large directional swings while staying long-only. A 20/50 EMA cross with a 10-day momentum confirmation enters on strength; exit on the opposite cross or when price breaks below the 50-day EMA. ATR-based trailing stop protects against sharp reversals. I'd change it if whipsaws dominate or if drawdowns exceed expectations, likely by adding a volatility regime filter or lengthening the slow EMA.
import math

FAST = 20
SLOW = 50
MOM = 10
ATR_N = 14
ATR_MULT = 3.0


def _ema(vals, n):
    if len(vals) < n:
        return None
    k = 2.0 / (n + 1.0)
    e = sum(vals[:n]) / n
    for v in vals[n:]:
        e = v * k + e * (1 - k)
    return e


def _atr(bars, n):
    highs = bars["high"]
    lows = bars["low"]
    closes = bars["close"]
    if len(closes) < n + 1:
        return None
    trs = []
    for i in range(len(closes) - n, len(closes)):
        tr = max(highs[i] - lows[i], abs(highs[i] - closes[i-1]), abs(lows[i] - closes[i-1]))
        trs.append(tr)
    return sum(trs) / len(trs)


def decide(bars, account):
    closes = bars["close"]
    n = len(closes)
    if n < SLOW + 2:
        return "HOLD"

    fast = _ema(closes, FAST)
    slow = _ema(closes, SLOW)
    if fast is None or slow is None:
        return "HOLD"

    price = closes[-1]
    mom = closes[-1] - closes[-1 - MOM] if n > MOM else 0.0
    atr = _atr(bars, ATR_N)

    pos = account.get("position", 0)
    entry = account.get("entry_price")

    if pos == 0:
        if fast > slow and mom > 0 and price > slow:
            return ("BUY", "EMA%d>EMA%d, mom>0, price>EMA%d" % (FAST, SLOW, SLOW))
        return "HOLD"
    else:
        if fast < slow:
            return ("SELL", "EMA%d<EMA%d bearish cross" % (FAST, SLOW))
        if price < slow:
            return ("SELL", "price below EMA%d" % SLOW)
        if entry is not None and atr is not None:
            if price < entry - ATR_MULT * atr:
                return ("SELL", "ATR trailing stop")
        return "HOLD"
