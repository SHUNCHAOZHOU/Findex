Trend-following with a fast/slow moving-average crossover plus a momentum filter, designed to capture TSLA's large directional swings while staying long-only. I use a 20/60 EMA crossover for entries and exits, with an ATR-based trailing stop to protect gains and reduce drawdowns. A volume/volatility sanity check avoids entering during chaotic spikes. I would change it if backtests show excessive whipsaw in sideways regimes, in which case I'd add a regime filter or lengthen the slow average.
import numpy as np

def _ema(values, span):
    alpha = 2.0 / (span + 1.0)
    out = [values[0]]
    for v in values[1:]:
        out.append(alpha * v + (1 - alpha) * out[-1])
    return out

def _atr(highs, lows, closes, period=14):
    n = len(closes)
    if n < 2:
        return 0.0
    trs = []
    for i in range(1, n):
        tr = max(highs[i] - lows[i], abs(highs[i] - closes[i-1]), abs(lows[i] - closes[i-1]))
        trs.append(tr)
    if len(trs) < period:
        return float(np.mean(trs)) if trs else 0.0
    return float(np.mean(trs[-period:]))

def decide(bars, account):
    closes = bars["close"]
    highs = bars["high"]
    lows = bars["low"]
    volumes = bars["volume"]
    n = len(closes)
    if n < 65:
        return "HOLD"

    fast = _ema(closes[-120:] if n >= 120 else closes, 20)
    slow = _ema(closes[-120:] if n >= 120 else closes, 60)
    f_now, s_now = fast[-1], slow[-1]
    f_prev, s_prev = fast[-2], slow[-2]

    price = closes[-1]
    atr = _atr(highs, lows, closes, 14)
    atr_pct = atr / price if price > 0 else 0.0

    # momentum: 10-day return
    mom = (closes[-1] / closes[-11] - 1.0) if n >= 11 else 0.0

    # volume sanity: current volume vs 20-day avg
    avg_vol = float(np.mean(volumes[-20:])) if n >= 20 else float(volumes[-1])
    vol_ratio = volumes[-1] / avg_vol if avg_vol > 0 else 1.0

    position = account.get("position", 0)
    entry = account.get("entry_price")

    # Exit logic
    if position > 0:
        # trailing stop: exit if price falls more than 2.5*ATR below entry or below slow EMA
        if entry is not None and price < entry - 2.5 * atr:
            return ("SELL", "trailing ATR stop hit")
        if f_now < s_now and f_prev >= s_prev:
            return ("SELL", "EMA cross down")
        if price < s_now * 0.97:
            return ("SELL", "price broke below slow EMA")
        return "HOLD"

    # Entry logic
    if f_now > s_now and f_prev <= s_prev:
        if mom > 0 and atr_pct < 0.08 and vol_ratio < 3.0:
            return ("BUY", "EMA cross up with momentum")
    # secondary: strong trend continuation
    if f_now > s_now and price > s_now and mom > 0.05 and atr_pct < 0.07:
        return ("BUY", "trend continuation")

    return "HOLD"
