import math

FAST = 10
SLOW = 30
TREND = 20
ATR_N = 14
ATR_MULT = 3.0
BREAKOUT_N = 10
PROFIT_ATR = 2.5


def _ema(vals, n):
    if len(vals) < n:
        return None
    k = 2.0 / (n + 1)
    e = sum(vals[:n]) / n
    for v in vals[n:]:
        e = v * k + e * (1 - k)
    return e


def _sma(vals, n):
    if len(vals) < n:
        return None
    return sum(vals[-n:]) / n


def _atr(bars, n):
    highs = bars["high"]
    lows = bars["low"]
    closes = bars["close"]
    if len(closes) < n + 1:
        return None
    trs = []
    for i in range(len(closes) - n, len(closes)):
        h = highs[i]
        l = lows[i]
        pc = closes[i - 1]
        tr = max(h - l, abs(h - pc), abs(l - pc))
        trs.append(tr)
    return sum(trs) / n


def decide(bars, account):
    closes = bars["close"]
    n = len(closes)
    if n < TREND + 6:
        return "HOLD"

    fast = _ema(closes, FAST)
    slow = _ema(closes, SLOW)
    trend_now = _sma(closes, TREND)
    trend_prev = _sma(closes[:-5], TREND) if n >= TREND + 5 else None
    price = closes[-1]

    if fast is None or slow is None or trend_now is None or trend_prev is None:
        return "HOLD"

    prior_high = max(closes[-(BREAKOUT_N + 1):-1])
    breakout = price > prior_high
    rising_trend = price > trend_now and trend_now > trend_prev
    uptrend = breakout or rising_trend

    downtrend = (fast < slow) and (price < trend_now) and (trend_now < trend_prev)

    pos = account.get("position", 0)
    entry = account.get("entry_price")

    if pos == 0:
        if uptrend:
            return ("BUY", "fast entry: breakout or rising 20d trend")
        return "HOLD"

    if downtrend:
        return ("SELL", "trend break")

    atr = _atr(bars, ATR_N)
    if atr is not None and entry is not None:
        if price < entry - ATR_MULT * atr:
            return ("SELL", "tight ATR stop")
        # profit-protecting exit: once meaningfully up, trail from highest close since entry
        if price > entry + PROFIT_ATR * atr:
            # find highest close since entry (approx: last 60 bars)
            look = min(n, 60)
            hi = max(closes[-look:])
            if price < hi - PROFIT_ATR * atr:
                return ("SELL", "profit trail stop")
            ema10 = _ema(closes, FAST)
            if ema10 is not None and price < ema10:
                return ("SELL", "close below 10d EMA while in profit")

    return "HOLD"
