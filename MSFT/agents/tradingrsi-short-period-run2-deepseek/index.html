import math

FAST = 20
SLOW = 60
REGIME = 100
ATR_N = 14
ATR_MULT = 5.0


def _ema(vals, n):
    if len(vals) < n:
        return None
    k = 2.0 / (n + 1.0)
    e = sum(vals[:n]) / n
    for v in vals[n:]:
        e = v * k + e * (1.0 - k)
    return e


def _sma(vals, n):
    if len(vals) < n:
        return None
    return sum(vals[-n:]) / n


def _atr(bars, n):
    highs = bars["high"]
    lows = bars["low"]
    closes = bars["close"]
    if len(closes) < n + 1:
        return None
    trs = []
    for i in range(len(closes) - n, len(closes)):
        tr = max(highs[i] - lows[i], abs(highs[i] - closes[i - 1]), abs(lows[i] - closes[i - 1]))
        trs.append(tr)
    return sum(trs) / len(trs)


def decide(bars, account):
    closes = bars["close"]
    if len(closes) < REGIME + 2:
        return "HOLD"

    fast = _ema(closes, FAST)
    slow = _ema(closes, SLOW)
    regime = _sma(closes, REGIME)
    atr = _atr(bars, ATR_N)
    price = closes[-1]

    if fast is None or slow is None or regime is None or atr is None:
        return "HOLD"

    pos = account.get("position", 0)
    entry = account.get("entry_price")

    if pos == 0:
        # Looser entry: participate in uptrends earlier.
        if fast > slow or price > regime:
            return ("BUY", "EMA%d>EMA%d or price>%dSMA" % (FAST, SLOW, REGIME))
        return "HOLD"

    # Exit only on a decisive regime break or a wide chandelier stop.
    if price < regime:
        return ("SELL", "price below %dSMA" % REGIME)

    if entry is not None:
        peak = max(closes[-min(len(closes), 120):])
        if price < peak - ATR_MULT * atr:
            return ("SELL", "chandelier stop hit")

    return "HOLD"

Trend-following with a fast/slow moving-average crossover plus a momentum filter, designed to capture sustained MSFT uptrends while exiting on trend breaks. I use a 20/60 EMA cross for entries and a 50-day SMA as a regime filter, with an ATR-based trailing stop from entry to protect gains. I'd change it if whipsaws dominate (raise smoothing) or if the stock becomes strongly mean-reverting (switch to a pullback/RSI approach).
import math

FAST = 20
SLOW = 60
TREND = 50
ATR_N = 14
ATR_MULT = 3.0


def _ema(vals, n):
    if len(vals) < n:
        return None
    k = 2.0 / (n + 1.0)
    e = sum(vals[:n]) / n
    for v in vals[n:]:
        e = v * k + e * (1.0 - k)
    return e


def _sma(vals, n):
    if len(vals) < n:
        return None
    return sum(vals[-n:]) / n


def _atr(bars, n):
    highs = bars["high"]
    lows = bars["low"]
    closes = bars["close"]
    if len(closes) < n + 1:
        return None
    trs = []
    for i in range(len(closes) - n, len(closes)):
        tr = max(highs[i] - lows[i], abs(highs[i] - closes[i - 1]), abs(lows[i] - closes[i - 1]))
        trs.append(tr)
    return sum(trs) / len(trs)


def decide(bars, account):
    closes = bars["close"]
    if len(closes) < SLOW + 2:
        return "HOLD"

    fast = _ema(closes, FAST)
    slow = _ema(closes, SLOW)
    trend = _sma(closes, TREND)
    atr = _atr(bars, ATR_N)
    price = closes[-1]

    if fast is None or slow is None or trend is None or atr is None:
        return "HOLD"

    pos = account.get("position", 0)
    entry = account.get("entry_price")

    if pos == 0:
        # Entry: fast above slow, price above trend, and short-term momentum positive
        if fast > slow and price > trend and closes[-1] > closes[-5]:
            return ("BUY", "EMA%d>EMA%d, price>%dSMA, momentum up" % (FAST, SLOW, TREND))
        return "HOLD"

    # Exit rules when holding
    if fast < slow:
        return ("SELL", "EMA%d crossed below EMA%d" % (FAST, SLOW))
    if price < trend:
        return ("SELL", "price below %dSMA" % TREND)
    if entry is not None and price < entry - ATR_MULT * atr:
        return ("SELL", "ATR trailing stop hit")
    return "HOLD"
