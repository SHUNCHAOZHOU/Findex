import math

FAST = 10
SLOW = 30
TREND = 100
ATR_N = 14
STOP_MULT = 4.0


def _sma(xs, n):
    if len(xs) < n:
        return None
    return sum(xs[-n:]) / n


def _atr(bars, n):
    highs = bars["high"]
    lows = bars["low"]
    closes = bars["close"]
    if len(closes) < n + 1:
        return None
    trs = []
    for i in range(len(closes) - n, len(closes)):
        tr = max(highs[i] - lows[i],
                 abs(highs[i] - closes[i - 1]),
                 abs(lows[i] - closes[i - 1]))
        trs.append(tr)
    return sum(trs) / n


def decide(bars, account):
    closes = bars["close"]
    n = len(closes)
    if n < SLOW + 2:
        return "HOLD"

    price = closes[-1]
    fast = _sma(closes, FAST)
    slow = _sma(closes, SLOW)
    trend = _sma(closes, TREND) if n >= TREND else None
    atr = _atr(bars, ATR_N)

    pos = account.get("position", 0)
    entry = account.get("entry_price")

    if pos > 0:
        # Exit only on a real trend break or a wide trailing stop.
        if trend is not None and price < trend:
            return ("SELL", "close below 100d SMA")
        if entry is not None and atr is not None and price < entry - STOP_MULT * atr:
            return ("SELL", "ATR trailing stop")
        return "HOLD"

    # Flat: look for entry. Loosened: no 6-day momentum or price>fast requirement.
    if fast is None or slow is None:
        return "HOLD"
    if fast <= slow:
        return "HOLD"
    if trend is not None and price < trend:
        return "HOLD"
    return ("BUY", "uptrend: fast>slow and above 100d SMA")

import math

FAST = 20
SLOW = 50
TREND = 200
ATR_N = 14
STOP_MULT = 3.0


def _sma(xs, n):
    if len(xs) < n:
        return None
    return sum(xs[-n:]) / n


def _atr(bars, n):
    highs = bars["high"]
    lows = bars["low"]
    closes = bars["close"]
    if len(closes) < n + 1:
        return None
    trs = []
    for i in range(len(closes) - n, len(closes)):
        tr = max(highs[i] - lows[i],
                 abs(highs[i] - closes[i - 1]),
                 abs(lows[i] - closes[i - 1]))
        trs.append(tr)
    return sum(trs) / n


def decide(bars, account):
    closes = bars["close"]
    n = len(closes)
    if n < SLOW + 2:
        return "HOLD"

    price = closes[-1]
    fast = _sma(closes, FAST)
    slow = _sma(closes, SLOW)
    trend = _sma(closes, TREND) if n >= TREND else None
    atr = _atr(bars, ATR_N)

    pos = account.get("position", 0)
    entry = account.get("entry_price")

    if pos > 0:
        # Exit on trend break or trailing stop from entry.
        if slow is not None and price < slow:
            return ("SELL", "close below 50d SMA")
        if trend is not None and price < trend:
            return ("SELL", "close below 200d SMA")
        if entry is not None and atr is not None and price < entry - STOP_MULT * atr:
            return ("SELL", "ATR trailing stop")
        return "HOLD"

    # Flat: look for entry.
    if fast is None or slow is None:
        return "HOLD"
    if fast <= slow:
        return "HOLD"
    if trend is not None and price < trend:
        return "HOLD"
    # Momentum confirmation: recent close above fast SMA and above 5-day-ago close.
    if n >= 6 and price > closes[-6] and price > fast:
        return ("BUY", "uptrend with momentum")
    return "HOLD"
