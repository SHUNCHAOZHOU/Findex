import math

FAST = 20
SLOW = 50
LONG = 200
ATR_N = 14
STOP_ATR = 3.0


def _sma(xs, n):
    if len(xs) < n:
        return None
    return sum(xs[-n:]) / n


def _atr(bars, n):
    highs = bars["high"]
    lows = bars["low"]
    closes = bars["close"]
    if len(closes) < n + 1:
        return None
    trs = []
    for i in range(len(closes) - n, len(closes)):
        tr = max(highs[i] - lows[i], abs(highs[i] - closes[i - 1]), abs(lows[i] - closes[i - 1]))
        trs.append(tr)
    return sum(trs) / n


def decide(bars, account):
    closes = bars["close"]
    n = len(closes)
    if n < SLOW + 2:
        return "HOLD"

    price = closes[-1]
    fast = _sma(closes, FAST)
    slow = _sma(closes, SLOW)
    long_ma = _sma(closes, LONG) if n >= LONG else None
    atr = _atr(bars, ATR_N)

    pos = account.get("position", 0)
    entry = account.get("entry_price")

    # Exit logic when holding
    if pos > 0:
        if entry is not None and atr is not None:
            if price < entry - STOP_ATR * atr:
                return ("SELL", "ATR trailing stop hit")
        if fast is not None and slow is not None and fast < slow:
            return ("SELL", "fast MA crossed below slow MA")
        if long_ma is not None and price < long_ma * 0.97:
            return ("SELL", "price well below 200-day MA")
        return "HOLD"

    # Entry logic when flat
    if fast is None or slow is None:
        return "HOLD"
    if long_ma is not None and price < long_ma:
        return "HOLD"
    # avoid buying into a huge one-day spike
    if n >= 2 and closes[-1] > closes[-2] * 1.06:
        return "HOLD"
    # primary: fast above slow
    if fast > slow:
        return ("BUY", "uptrend: fast MA above slow MA")
    # recovery: 50-day MA rising and price above it
    slow_prev = _sma(closes[:-5], SLOW) if n >= SLOW + 5 else None
    if slow_prev is not None and slow > slow_prev and price > slow:
        return ("BUY", "recovery: rising 50-day MA")
    return "HOLD"

import math

FAST = 20
SLOW = 50
LONG = 200
ATR_N = 14
STOP_ATR = 3.0


def _sma(xs, n):
    if len(xs) < n:
        return None
    return sum(xs[-n:]) / n


def _atr(bars, n):
    highs = bars["high"]
    lows = bars["low"]
    closes = bars["close"]
    if len(closes) < n + 1:
        return None
    trs = []
    for i in range(len(closes) - n, len(closes)):
        tr = max(highs[i] - lows[i], abs(highs[i] - closes[i - 1]), abs(lows[i] - closes[i - 1]))
        trs.append(tr)
    return sum(trs) / n


def decide(bars, account):
    closes = bars["close"]
    n = len(closes)
    if n < SLOW + 2:
        return "HOLD"

    price = closes[-1]
    fast = _sma(closes, FAST)
    slow = _sma(closes, SLOW)
    long_ma = _sma(closes, LONG) if n >= LONG else None
    atr = _atr(bars, ATR_N)

    pos = account.get("position", 0)
    entry = account.get("entry_price")

    # Exit logic when holding
    if pos > 0:
        if entry is not None and atr is not None:
            if price < entry - STOP_ATR * atr:
                return ("SELL", "ATR trailing stop hit")
        if fast is not None and slow is not None and fast < slow:
            return ("SELL", "fast MA crossed below slow MA")
        if long_ma is not None and price < long_ma * 0.97:
            return ("SELL", "price well below 200-day MA")
        return "HOLD"

    # Entry logic when flat
    if fast is None or slow is None:
        return "HOLD"
    if fast <= slow:
        return "HOLD"
    if long_ma is not None and price < long_ma:
        return "HOLD"
    # momentum: price above 20-day MA and recent 5-day return positive
    if price < fast:
        return "HOLD"
    if n >= 6 and closes[-1] <= closes[-6]:
        return "HOLD"
    # avoid buying into a huge one-day spike
    if n >= 2 and closes[-1] > closes[-2] * 1.06:
        return "HOLD"
    return ("BUY", "uptrend with positive momentum")
