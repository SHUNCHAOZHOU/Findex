import statistics

SMA_LEN = 50
MOM_LEN = 20
BUY_BUFFER = 0.005
SELL_BUFFER = 0.01


def _sma(vals, n):
    if len(vals) < n:
        return None
    return sum(vals[-n:]) / n


def decide(bars, account):
    close = bars["close"]
    n = len(close)
    if n < SMA_LEN + 2:
        return "HOLD"

    sma = _sma(close, SMA_LEN)
    price = close[-1]

    mom = None
    if n > MOM_LEN:
        past = close[-1 - MOM_LEN]
        if past > 0:
            mom = price / past - 1.0

    holding = account.get("position", 0) > 0

    if not holding:
        if price > sma * (1.0 + BUY_BUFFER) and (mom is None or mom > 0):
            return ("BUY", "price above SMA50 with positive momentum")
        return "HOLD"
    else:
        if price < sma * (1.0 - SELL_BUFFER) or (mom is not None and mom < -0.07):
            return ("SELL", "trend broken: price below SMA50 or momentum negative")
        return "HOLD"

Trend-following long-only strategy for CVX: buy when price is above a rising 50-day SMA and momentum (20-day return) is positive, confirming an uptrend; exit when price closes below the 50-day SMA or momentum turns negative. This captures sustained energy-sector uptrends while cutting losses in downtrends. I'd change it if it whipsaws too much (add a buffer/confirmation band) or if CVX becomes strongly mean-reverting.
import statistics

SMA_LEN = 50
MOM_LEN = 20
BUFFER = 0.005


def _sma(vals, n):
    if len(vals) < n:
        return None
    return sum(vals[-n:]) / n


def decide(bars, account):
    close = bars["close"]
    n = len(close)
    if n < SMA_LEN + 2:
        return "HOLD"

    sma = _sma(close, SMA_LEN)
    sma_prev = _sma(close[:-1], SMA_LEN)
    price = close[-1]

    mom = None
    if n > MOM_LEN:
        past = close[-1 - MOM_LEN]
        if past > 0:
            mom = price / past - 1.0

    holding = account.get("position", 0) > 0

    if not holding:
        if price > sma * (1.0 + BUFFER) and sma > sma_prev and (mom is None or mom > 0):
            return ("BUY", "uptrend: price above rising SMA50 with positive momentum")
        return "HOLD"
    else:
        if price < sma * (1.0 - BUFFER) or (mom is not None and mom < -0.05):
            return ("SELL", "trend broken: price below SMA50 or momentum negative")
        return "HOLD"
