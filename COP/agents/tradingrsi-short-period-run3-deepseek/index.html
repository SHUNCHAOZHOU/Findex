import numpy as np

FAST = 10
SLOW = 30
TREND = 100
ATR_N = 14
ATR_STOP = 3.0


def _sma(x, n):
    if len(x) < n:
        return None
    return float(np.mean(x[-n:]))


def _atr(bars, n):
    high = bars["high"]
    low = bars["low"]
    close = bars["close"]
    if len(close) < n + 1:
        return None
    trs = []
    for i in range(len(close) - n, len(close)):
        tr = max(high[i] - low[i], abs(high[i] - close[i - 1]), abs(low[i] - close[i - 1]))
        trs.append(tr)
    return float(np.mean(trs))


def decide(bars, account):
    close = bars["close"]
    n = len(close)
    if n < SLOW + 2:
        return "HOLD"

    price = close[-1]
    sma_fast = _sma(close, FAST)
    sma_slow = _sma(close, SLOW)
    sma_trend = _sma(close, TREND) if n >= TREND else None
    sma_slow_prev = _sma(close[:-5], SLOW) if n >= SLOW + 5 else None

    pos = account.get("position", 0)
    entry = account.get("entry_price")

    if pos > 0:
        # Exit on trend break
        if sma_slow is not None and price < sma_slow:
            return ("SELL", "close below 30d SMA")
        # ATR-based trailing stop from entry
        atr = _atr(bars, ATR_N)
        if entry is not None and atr is not None and price < entry - ATR_STOP * atr:
            return ("SELL", "ATR trailing stop hit")
        return "HOLD"

    # Flat: look for entry
    if sma_fast is None or sma_slow is None:
        return "HOLD"
    if sma_fast <= sma_slow:
        return "HOLD"
    if price <= sma_slow:
        return "HOLD"
    if sma_trend is not None and price < sma_trend:
        return "HOLD"
    # require the slow average to be rising to avoid buying into a downtrend bounce
    if sma_slow_prev is not None and sma_slow <= sma_slow_prev:
        return "HOLD"
    return ("BUY", "uptrend: price>30d, 10d>30d, rising 30d, above 100d")

Trend-following long-only strategy for an energy major: buy when price is above a rising 50-day average and momentum is positive, exit when price breaks below the 50-day average or a trailing stop from the entry is hit. This captures sustained uptrends while cutting losses in downtrends, which suits a cyclical stock like COP. import numpy as np

FAST = 20
SLOW = 50
TREND = 200
STOP = 0.10


def _sma(x, n):
    if len(x) < n:
        return None
    return float(np.mean(x[-n:]))


def decide(bars, account):
    close = bars["close"]
    n = len(close)
    if n < SLOW + 2:
        return "HOLD"

    price = close[-1]
    sma_fast = _sma(close, FAST)
    sma_slow = _sma(close, SLOW)
    sma_trend = _sma(close, TREND) if n >= TREND else None

    pos = account.get("position", 0)
    entry = account.get("entry_price")

    if pos > 0:
        # Exit on trend break or trailing stop
        if sma_slow is not None and price < sma_slow:
            return ("SELL", "close below 50d SMA")
        if entry is not None and price < entry * (1.0 - STOP):
            return ("SELL", "trailing stop hit")
        return "HOLD"

    # Flat: look for entry
    if sma_fast is None or sma_slow is None:
        return "HOLD"
    if sma_fast <= sma_slow:
        return "HOLD"
    if price <= sma_slow:
        return "HOLD"
    if sma_trend is not None and price < sma_trend:
        return "HOLD"
    # momentum confirmation: 10-day return positive
    if n >= 11 and close[-1] <= close[-11]:
        return "HOLD"
    return ("BUY", "uptrend: price>50d, 20d>50d, positive momentum")
