import numpy as np

FAST = 50
SLOW = 200
TRAIL = 0.92


def _sma(arr, n):
    if len(arr) < n:
        return None
    return float(np.mean(arr[-n:]))


def decide(bars, account):
    closes = bars["close"]
    n = len(closes)
    if n < SLOW + 2:
        return "HOLD"

    price = closes[-1]
    fast = _sma(closes, FAST)
    slow = _sma(closes, SLOW)

    if fast is None or slow is None:
        return "HOLD"

    pos = account.get("position", 0)
    entry = account.get("entry_price")

    uptrend = fast > slow and price > slow
    downtrend = fast < slow and price < slow

    if pos == 0:
        if uptrend:
            return ("BUY", "uptrend")
        return "HOLD"
    else:
        if downtrend:
            return ("SELL", "trend breakdown")
        if entry is not None:
            peak = max(closes[-60:])
            if price < peak * TRAIL and price < entry:
                return ("SELL", "trailing stop")
        return "HOLD"

import numpy as np

FAST = 50
SLOW = 200
RSI_N = 14
STOP = 0.88


def _sma(arr, n):
    if len(arr) < n:
        return None
    return float(np.mean(arr[-n:]))


def _rsi(closes, n=RSI_N):
    if len(closes) < n + 1:
        return None
    diffs = np.diff(closes[-(n + 1):])
    gains = np.where(diffs > 0, diffs, 0.0)
    losses = np.where(diffs < 0, -diffs, 0.0)
    ag = gains.mean()
    al = losses.mean()
    if al == 0:
        return 100.0
    rs = ag / al
    return 100.0 - 100.0 / (1.0 + rs)


def decide(bars, account):
    closes = bars["close"]
    n = len(closes)
    if n < SLOW + 2:
        return "HOLD"

    price = closes[-1]
    fast = _sma(closes, FAST)
    slow = _sma(closes, SLOW)
    rsi = _rsi(closes)

    if fast is None or slow is None or rsi is None:
        return "HOLD"

    pos = account.get("position", 0)
    entry = account.get("entry_price")

    uptrend = fast > slow and price > slow
    downtrend = fast < slow and price < slow

    if pos == 0:
        # Enter on a confirmed long-term uptrend; avoid chasing overbought spikes.
        if uptrend and rsi < 72:
            return ("BUY", "uptrend, RSI %.0f" % rsi)
        return "HOLD"
    else:
        # Exit only on a genuine long-term trend break or a wide stop.
        if downtrend:
            return ("SELL", "trend breakdown")
        if entry is not None and price < entry * STOP:
            return ("SELL", "stop loss")
        return "HOLD"

This is a trend-following strategy for CVX that combines a 50/200-day moving-average regime filter with a shorter-term momentum/mean-reversion entry. It buys when price is above the long-term trend (50-day MA above 200-day MA) and short-term momentum turns positive (close crosses above a 20-day MA or RSI recovers from oversold), and sells when the trend breaks down (close below 200-day MA or 50-day crosses below 200-day) or RSI becomes overbought with price extended. I'd change it if it underperforms buy-and-hold over a long stretch, or if whipsawing in choppy markets causes excessive churn.
import numpy as np

FAST = 50
SLOW = 200
MID = 20
RSI_N = 14

def _sma(arr, n):
    if len(arr) < n:
        return None
    return float(np.mean(arr[-n:]))

def _rsi(closes, n=RSI_N):
    if len(closes) < n + 1:
        return None
    diffs = np.diff(closes[-(n + 1):])
    gains = np.where(diffs > 0, diffs, 0.0)
    losses = np.where(diffs < 0, -diffs, 0.0)
    ag = gains.mean()
    al = losses.mean()
    if al == 0:
        return 100.0
    rs = ag / al
    return 100.0 - 100.0 / (1.0 + rs)

def decide(bars, account):
    closes = bars["close"]
    n = len(closes)
    if n < SLOW + 2:
        return "HOLD"

    price = closes[-1]
    fast = _sma(closes, FAST)
    slow = _sma(closes, SLOW)
    mid = _sma(closes, MID)
    rsi = _rsi(closes)

    if fast is None or slow is None or mid is None or rsi is None:
        return "HOLD"

    pos = account.get("position", 0)
    entry = account.get("entry_price")

    uptrend = fast > slow and price > slow
    downtrend = fast < slow or price < slow * 0.97

    if pos == 0:
        # Entry: uptrend regime + short-term momentum confirmation
        if uptrend and price > mid and 40 < rsi < 75:
            return ("BUY", "uptrend, price above 20d MA, RSI %.0f" % rsi)
        # Deep oversold bounce within longer uptrend
        if fast > slow and rsi < 32 and price > slow:
            return ("BUY", "oversold bounce in uptrend, RSI %.0f" % rsi)
        return "HOLD"
    else:
        # Exit conditions
        if downtrend:
            return ("SELL", "trend breakdown")
        if rsi > 80 and price > mid * 1.08:
            return ("SELL", "overbought extension")
        if entry is not None and price < entry * 0.92:
            return ("SELL", "stop loss")
        return "HOLD"
