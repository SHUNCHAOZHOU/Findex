import statistics

FAST = 20
SLOW = 50
LONG = 100

def _sma(vals, n):
    if len(vals) < n:
        return None
    return sum(vals[-n:]) / n

def decide(bars, account):
    close = bars["close"]
    n = len(close)
    if n < LONG + 5:
        return "HOLD"
    price = close[-1]
    sma_fast = _sma(close, FAST)
    sma_slow = _sma(close, SLOW)
    sma_long = _sma(close, LONG)
    pos = account.get("position", 0)

    if pos == 0:
        # Enter when price is above the long average and the fast average is
        # above the slow one (uptrend regime). Permissive: no momentum gate.
        if price > sma_long and sma_fast > sma_slow:
            return ("BUY", "uptrend: price>100sma and 20>50sma")
        return "HOLD"
    else:
        # Exit only on a decisive break: below long average AND fast below slow.
        if price < sma_long and sma_fast < sma_slow:
            return ("SELL", "regime break: below 100sma and 20<50sma")
        return "HOLD"

Trend-following with a 50/200-day moving-average regime filter plus a faster 20-day momentum trigger, designed to capture COP's cyclical energy swings while staying long-only. I buy when price is above the 200-day average and the 20-day average is rising with positive short-term momentum, and exit on a break below the 50-day average or a sharp momentum reversal. import statistics

FAST = 20
SLOW = 50
LONG = 200
MOM = 10

def _sma(vals, n):
    if len(vals) < n:
        return None
    return sum(vals[-n:]) / n

def decide(bars, account):
    close = bars["close"]
    n = len(close)
    if n < LONG + 5:
        return "HOLD"
    price = close[-1]
    sma_fast = _sma(close, FAST)
    sma_slow = _sma(close, SLOW)
    sma_long = _sma(close, LONG)
    prev_fast = _sma(close[:-5], FAST)
    mom = price - close[-MOM - 1]
    pos = account.get("position", 0)

    if pos == 0:
        if (price > sma_long and sma_fast > sma_slow and
                prev_fast is not None and sma_fast > prev_fast and mom > 0):
            return ("BUY", "uptrend: price>200sma, 20>50sma rising, positive momentum")
        return "HOLD"
    else:
        if price < sma_slow or mom < 0 and price < sma_fast:
            return ("SELL", "trend break: below 50sma or momentum reversal")
        return "HOLD"
