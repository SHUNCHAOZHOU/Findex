import numpy as np

FAST = 10
SLOW = 30
RSI_N = 14
ATR_N = 14
ATR_STOP_MULT = 3.5
RSI_MAX = 80.0
RSI_MIN = 40.0


def _sma(x, n):
    if len(x) < n:
        return None
    return float(np.mean(x[-n:]))


def _rsi(close, n):
    if len(close) < n + 1:
        return None
    c = np.asarray(close[-(n + 1):], dtype=float)
    d = np.diff(c)
    up = np.where(d > 0, d, 0.0)
    dn = np.where(d < 0, -d, 0.0)
    au = up.mean()
    ad = dn.mean()
    if ad == 0:
        return 100.0
    rs = au / ad
    return 100.0 - 100.0 / (1.0 + rs)


def _atr(high, low, close, n):
    if len(close) < n + 1:
        return None
    h = np.asarray(high[-(n + 1):], dtype=float)
    l = np.asarray(low[-(n + 1):], dtype=float)
    c = np.asarray(close[-(n + 1):], dtype=float)
    tr = np.maximum(h[1:] - l[1:], np.maximum(np.abs(h[1:] - c[:-1]), np.abs(l[1:] - c[:-1])))
    return float(tr.mean())


def decide(bars, account):
    close = bars["close"]
    high = bars["high"]
    low = bars["low"]
    n = len(close)
    if n < SLOW + 2:
        return "HOLD"

    price = close[-1]
    fast = _sma(close, FAST)
    slow = _sma(close, SLOW)
    rsi = _rsi(close, RSI_N)
    atr = _atr(high, low, close, ATR_N)
    fast_prev = _sma(close[:-1], FAST)

    pos = account.get("position", 0)
    entry = account.get("entry_price")

    if pos > 0:
        if fast is not None and slow is not None and fast < slow and price < slow:
            return ("SELL", "fast MA below slow MA and price below slow")
        if entry is not None and atr is not None:
            if price < entry - ATR_STOP_MULT * atr:
                return ("SELL", "ATR stop hit")
        return "HOLD"

    if fast is None or slow is None or rsi is None:
        return "HOLD"

    uptrend = fast > slow and price > slow
    rsi_ok = RSI_MIN < rsi < RSI_MAX
    if uptrend and rsi_ok:
        return ("BUY", "uptrend with RSI in healthy band")

    # Secondary momentum entry: strong trend, allow elevated RSI
    if fast_prev is not None and fast > fast_prev and price > slow and rsi < 88.0:
        return ("BUY", "rising fast MA above slow, momentum entry")

    return "HOLD"

import numpy as np

FAST = 20
SLOW = 50
LONG = 200
RSI_N = 14
ATR_N = 14
ATR_STOP_MULT = 3.0


def _sma(x, n):
    if len(x) < n:
        return None
    return float(np.mean(x[-n:]))


def _rsi(close, n):
    if len(close) < n + 1:
        return None
    c = np.asarray(close[-(n + 1):], dtype=float)
    d = np.diff(c)
    up = np.where(d > 0, d, 0.0)
    dn = np.where(d < 0, -d, 0.0)
    au = up.mean()
    ad = dn.mean()
    if ad == 0:
        return 100.0
    rs = au / ad
    return 100.0 - 100.0 / (1.0 + rs)


def _atr(high, low, close, n):
    if len(close) < n + 1:
        return None
    h = np.asarray(high[-(n + 1):], dtype=float)
    l = np.asarray(low[-(n + 1):], dtype=float)
    c = np.asarray(close[-(n + 1):], dtype=float)
    tr = np.maximum(h[1:] - l[1:], np.maximum(np.abs(h[1:] - c[:-1]), np.abs(l[1:] - c[:-1])))
    return float(tr.mean())


def decide(bars, account):
    close = bars["close"]
    high = bars["high"]
    low = bars["low"]
    n = len(close)
    if n < SLOW + 2:
        return "HOLD"

    price = close[-1]
    fast = _sma(close, FAST)
    slow = _sma(close, SLOW)
    long_ma = _sma(close, LONG) if n >= LONG else None
    rsi = _rsi(close, RSI_N)
    atr = _atr(high, low, close, ATR_N)

    pos = account.get("position", 0)
    entry = account.get("entry_price")

    if pos > 0:
        # Exit on trend break or ATR trailing stop from entry
        if fast is not None and slow is not None and fast < slow:
            return ("SELL", "fast MA below slow MA")
        if entry is not None and atr is not None:
            if price < entry - ATR_STOP_MULT * atr:
                return ("SELL", "ATR trailing stop hit")
        if long_ma is not None and price < long_ma * 0.97:
            return ("SELL", "below long MA buffer")
        return "HOLD"

    # Flat: look for entry
    if fast is None or slow is None or rsi is None:
        return "HOLD"
    uptrend = fast > slow and price > slow
    if long_ma is not None:
        uptrend = uptrend and price > long_ma
    not_overbought = rsi < 72
    momentum = price > close[-FAST] if n > FAST else False

    if uptrend and not_overbought and momentum:
        return ("BUY", "uptrend with momentum, RSI ok")
    return "HOLD"
