import statistics

FAST = 20
EXIT = 50
MOM = 20
DRAWDOWN = 0.10


def _sma(vals, n):
    if len(vals) < n:
        return None
    return sum(vals[-n:]) / n


def decide(bars, account):
    close = bars["close"]
    n = len(close)
    if n < EXIT + 5:
        return "HOLD"

    price = close[-1]
    sma_fast = _sma(close, FAST)
    sma_fast_prev = sum(close[-FAST-1:-1]) / FAST
    sma_exit = _sma(close, EXIT)

    mom = price / close[-MOM-1] - 1.0 if n > MOM else 0.0

    recent_high = max(close[-60:]) if n >= 60 else max(close)
    dd = price / recent_high - 1.0

    # Early entry: fresh cross above 20d SMA, or strong momentum above 50d SMA
    cross_up = price > sma_fast and close[-2] <= sma_fast_prev
    strong_mom = mom > 0.02 and price > sma_exit
    entry = cross_up or strong_mom

    if account["position"] == 0:
        if entry and dd > -DRAWDOWN:
            return ("BUY", "early cross or strong momentum")
        return "HOLD"
    else:
        if price < sma_exit or dd < -DRAWDOWN:
            return ("SELL", "trend break or drawdown")
        return "HOLD"

import statistics

FAST = 50
SLOW = 200
MOM = 20
DRAWDOWN = 0.12


def _sma(vals, n):
    if len(vals) < n:
        return None
    return sum(vals[-n:]) / n


def decide(bars, account):
    close = bars["close"]
    n = len(close)
    if n < SLOW + 5:
        return "HOLD"

    price = close[-1]
    sma_fast = _sma(close, FAST)
    sma_slow = _sma(close, SLOW)
    sma_fast_prev = sum(close[-FAST-5:-5]) / FAST

    mom = price / close[-MOM-1] - 1.0 if n > MOM else 0.0

    recent_high = max(close[-60:]) if n >= 60 else max(close)
    dd = price / recent_high - 1.0

    uptrend = price > sma_slow and sma_fast > sma_slow and sma_fast > sma_fast_prev
    momentum_ok = mom > 0.0

    if account["position"] == 0:
        if uptrend and momentum_ok and dd > -DRAWDOWN:
            return ("BUY", "uptrend+momentum")
        return "HOLD"
    else:
        if price < sma_fast or dd < -DRAWDOWN:
            return ("SELL", "trend break or drawdown")
        return "HOLD"
