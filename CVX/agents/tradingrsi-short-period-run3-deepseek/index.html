import statistics

SLOW = 50
LONG = 200
EXIT_BUF = 0.02

def _sma(vals, n):
    if len(vals) < n:
        return None
    return sum(vals[-n:]) / n

def decide(bars, account):
    close = bars["close"]
    n = len(close)
    if n < SLOW + 2:
        return "HOLD"

    price = close[-1]
    sma_slow = _sma(close, SLOW)
    sma_slow_prev = sum(close[-SLOW-1:-1]) / SLOW
    sma_long = _sma(close, LONG)

    holding = account.get("position", 0) > 0

    rising = sma_slow > sma_slow_prev
    above_slow = price > sma_slow
    long_ok = (sma_long is None) or (price > sma_long)

    if not holding:
        if above_slow and rising and long_ok:
            return ("BUY", "above rising 50SMA")
        return "HOLD"
    else:
        if price < sma_slow * (1.0 - EXIT_BUF):
            return ("SELL", "close below 50SMA buffer")
        return "HOLD"

Trend-following long-only strategy for CVX: buy when price is above a rising 50-day SMA and momentum (20-day return) is positive, confirming an uptrend; exit when price closes below the 50-day SMA or momentum turns negative. This captures sustained oil-major uptrends while avoiding deep drawdowns, and I'd revisit if whipsaws cause excessive churn or if the stock becomes strongly mean-reverting.
import statistics

FAST = 20
SLOW = 50
MOM = 20

def _sma(vals, n):
    if len(vals) < n:
        return None
    return sum(vals[-n:]) / n

def decide(bars, account):
    close = bars["close"]
    n = len(close)
    if n < SLOW + 2:
        return "HOLD"

    sma_slow = _sma(close, SLOW)
    sma_slow_prev = sum(close[-SLOW-1:-1]) / SLOW
    sma_fast = _sma(close, FAST)
    price = close[-1]

    mom = None
    if n > MOM:
        past = close[-MOM-1]
        if past > 0:
            mom = price / past - 1.0

    holding = account.get("position", 0) > 0

    uptrend = price > sma_slow and sma_slow > sma_slow_prev
    momentum_ok = mom is not None and mom > 0
    fast_ok = sma_fast is not None and price > sma_fast

    if not holding:
        if uptrend and momentum_ok and fast_ok:
            return ("BUY", "uptrend+momentum")
        return "HOLD"
    else:
        if price < sma_slow or (mom is not None and mom < -0.02):
            return ("SELL", "trend break")
        return "HOLD"
